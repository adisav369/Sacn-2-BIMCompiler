```python
def recognize_2d_to_db(svg_drawing):
    # Parse SVG into primitive elements
    primitives = parse_svg(svg_drawing)  # lines, arcs, text, hatches
    
    # Group primitives into probable elements using mapping rules
    candidates = []
    for primitive in primitives:
        # Which rule matches this primitive's style?
        rule = find_matching_rule(primitive.style, mapping_rules)
        
        # What element type does the rule suggest?
        element_type = rule.suggested_ifc_class
        
        # What's the geometric context?
        context = analyze_neighbors(primitive, primitives)
        
        candidates.append({
            'type': element_type,
            'geometry': primitive.bounds,
            'confidence': rule.confidence(context),
            'rule_id': rule.id
        })
    
    # Use spatial reasoning to resolve overlaps and gaps
    resolved = spatial_resolver(candidates)
    
    # Generate DB records
    for element in resolved:
        db.insert_element({
            'ifc_class': element.type,
            'minX': element.bounds.xmin,
            'maxX': element.bounds.xmax,
            # ... other fields
            'source_2d_id': element.id,
            'recognition_confidence': element.confidence
        })
    
    return db
```

```python
class DrawingMapper:
    def __init__(self):
        # Learned from Synthetic Rosetta Stone
        self.element_to_2d = {
            'IfcWall': self.wall_to_2d,
            'IfcDoor': self.door_to_2d,
            'IfcWindow': self.window_to_2d,
            'IfcSlab': self.slab_to_2d,
            'IfcFurniture': self.furniture_to_2d,
        }
        
        self._2d_to_element = {
            'closed_polygon_with_hatch': self.hatched_polygon_to_element,
            'arc_with_line': self.swing_arc_to_door,
            'rectangle_with_cross': self.window_symbol_to_window,
            'text_near_symbol': self.annotation_to_property,
        }
    
    def db_to_2d(self, element, view_type):
        """Generate 2D representation from DB element"""
        if view_type == 'plan' and element.ifc_class in self.element_to_2d:
            return self.element_to_2d[element.ifc_class](element, 'plan')
        # ... other view types
    
    def _2d_to_db(self, primitives, context):
        """Recognize DB elements from 2D primitives"""
        candidates = []
        for primitive in primitives:
            for pattern, recognizer in self._2d_to_element.items():
                if primitive.matches(pattern):
                    element = recognizer(primitive, context)
                    candidates.append(element)
        return self.resolve_conflicts(candidates)
```

```python
class TopologyMaker:
    def __init__(self, library: ComponentLibrary):
        self.library = library
        self.verb_stack = []
        
    def combine(self, base_id, additions, operation="assembly"):
        # Like Photoshop's "Merge Layers" but parametric
        base = self.library.get_component(base_id)
        parts = [self.library.get_component(a) for a in additions]
        
        # Generate new geometry
        if operation == "boolean_merge":
            result_geo = boolean_union(base.geometry, [p.geometry for p in parts])
        elif operation == "assembly_group":
            result_geo = group_components([base] + parts)  # Keep separate
        
        # Merge ERP attributes (with conflict resolution)
        result_erp = self._merge_erp_attributes([base] + parts)
        
        # Generate Java verb for this operation
        verb = self._create_verb("COMBINE", base_id, additions, operation)
        self.verb_stack.append(verb)
        
        return {
            "geometry": result_geo,
            "erp_attributes": result_erp,
            "verb_stack": self.verb_stack
        }
    
    def apply_theme(self, component_id, theme_name):
```

### 3. `VerbEngine.java` — The Router
```java
// The only coordinator that matters
public class VerbEngine {
    private final Map<String, VerbHandler> handlers = new ConcurrentHashMap<>();
    private final List<VerbListener> listeners = new CopyOnWriteArrayList<>();
    
    // Register a handler for an action
    public void register(String action, VerbHandler handler) {
        handlers.put(action, handler);
    }
    
    // Execute a verb
    public VerbHandler.Result execute(Verb verb) {
        // Notify listeners before execution
        listeners.forEach(l -> l.before(verb));
        
        long start = System.nanoTime();
        try {
            VerbHandler handler = handlers.get(verb.action);
            if (handler == null) {
                return VerbHandler.Result.error("Unknown action: " + verb.action);
            }
            
            VerbHandler.Result result = handler.handle(verb);
            
            // Notify listeners after execution
            listeners.forEach(l -> l.after(verb, result, System.nanoTime() - start));
            
            return result;
        } catch (Exception e) {
            VerbHandler.Result error = VerbHandler.Result.error(e.getMessage());
            listeners.forEach(l -> l.after(verb, error, System.nanoTime() - start));
            return error;
        }
    }
    
    // Listeners for logging, metrics, history
    public interface VerbListener {
        default void before(Verb verb) {}
        default void after(Verb verb, VerbHandler.Result result, long durationNs) {}
    }
    
    public void addListener(VerbListener listener) {
        listeners.add(listener);
    }
}
```

**Why this is enough:** Simple map lookup. No complex routing, no dependency injection, no XML config. Just code.

### 5. `Bridge.py` — The Blender Connection
```python
# The only Python that matters
import bpy
import json
import socket
import threading
from collections import deque

class VerbBridge:
    def __init__(self, host='localhost', port=9999):
        self.host = host
        self.port = port
        self.socket = None
        self.pending_results = deque()
        self.verb_queue = deque()
        
        # Start connection thread
        self.connect()
        
        # Register timer for main thread updates
        bpy.app.timers.register(self.process_pending, first_interval=0.1)
    
    def connect(self):
        """Connect to Java backend"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        
        # Start receiver thread
        threading.Thread(target=self.receive_loop, daemon=True).start()
    
    def send_verb(self, action, target=None, params=None):
        """Send a verb to Java backend"""
        verb = {
            'action': action,
            'target': target,
            'params': params or {},
            'user': bpy.context.scene.get('user', 'unknown'),
            'timestamp': time.time()
        }
        
        # Send as JSON
        self.socket.sendall((json.dumps(verb) + '\n').encode())
    
    def receive_loop(self):
        """Receive results in background thread"""
        buffer = ''
        while True:
            data = self.socket.recv(4096).decode()
            if not data:
                break
            
            buffer += data
            while '\n' in buffer:
                line, buffer = buffer.split('\n', 1)
                result = json.loads(line)
                self.pending_results.append(result)  # Thread-safe deque
    
    def process_pending(self):
        """Process results on main thread (called by timer)"""
        while self.pending_results:
            result = self.pending_results.popleft()
            self.apply_to_viewport(result)
        
        return 0.1  # Keep timer alive
    
    def apply_to_viewport(self, result):
        """Update Blender viewport with result"""
        if 'geometry' in result:

            if result.get('global_id') in bpy.data.objects:

                # Create new object
                self.create_object(result)

                if area.type == 'VIEW_3D':
                    area.tag_redraw()

bridge = VerbBridge()

# Decorator for Blender operators to auto-send verbs
def verb_action(action):
    def decorator(cls):
        original_execute = cls.execute
        
        def wrapped_execute(self, context):
            # Capture parameters from UI
            params = {k: getattr(self, k) for k in dir(self) 
                     if not k.startswith('_') and not callable(getattr(self, k))}
            
            # Send verb before executing
            bridge.send_verb(action, context.active_object.name if context.active_object else None, params)
            
            # Execute original
            return original_execute(self, context)
        
        cls.execute = wrapped_execute
        return cls
    return decorator

# Example usage

class BIM_OT_create_wall(bpy.types.Operator):

        # Bonsai's actual wall creation code
        bpy.ops.bim.add_wall(type=self.wall_type, height=self.height)
```

## 🎨 The Blender Viewport: Just a Renderer

Your Python bridge becomes even thinner—it just:

1. **Subscribes to DB changes** (via websocket or polling)
2. **Renders whatever the DB says** exists
3. **Sends user actions** as verbs to Java
4. **Applies local view filters** without touching DB

```python
class BlenderViewport:
    def __init__(self, db_connection):
        self.db = db_connection
        self.local_filters = []  # View-only filters
        self.selection = SelectionManager()
        self.layers = LayerManager()
        
        # Subscribe to DB changes
        self.db.subscribe(self.on_database_change)
        
        # Start render loop
        bpy.app.timers.register(self.render_frame, first_interval=1/60)  # 60fps
    
    def on_database_change(self, changes):
        """DB changed (by another user or Java backend)"""
        for change in changes:
            if change.type == 'element_created':
                self.add_element(change.element)
            elif change.type == 'element_modified':
                self.update_element(change.element)
            elif change.type == 'element_deleted':
                self.remove_element(change.element_id)
    
    def render_frame(self):
        """Called 60x per second"""
        # Apply local filters (color by status, hide layers, etc.)
        filtered_elements = self.apply_filters(self.db.current_elements)
        
        # Update viewport
        self.update_geometry(filtered_elements)
        
        # Draw selection UI
        self.draw_selection_overlay()
        
        return 1/60  # Keep timer
    
    def on_user_action(self, action, params):
        """User did something in Blender"""
        # Convert to verb
        verb = Verb(
            action=action,
            target=self.selection.active if self.selection.active else None,
            params=params,
            user=bpy.context.scene.get('user'),
            timestamp=time.time()
        )
        
        # Send to Java backend (not direct to DB!)
        self.java_backend.execute(verb)
        
        # Optimistic local update (optional)
        self.optimistic_update(verb)
```

## 🔄 The Undo/Redo Magic

```java
// When user does something:
public class UndoManager {
    private Stack<Verb> undoStack = new Stack<>();
    private Stack<Verb> redoStack = new Stack<>();
    
    public void execute(Verb verb) {
        // 1. Generate inverse verb
        Verb inverse = generateInverse(verb);
        
        // 2. Execute original
        Result result = engine.execute(verb);
        
        // 3. Store for undo
        undoStack.push(inverse);
        redoStack.clear();
        
        // 4. Broadcast to all viewports
        broadcast(verb);
    }
    
    public void undo() {
        if (!undoStack.isEmpty()) {
            Verb inverse = undoStack.pop();
            Result result = engine.execute(inverse);
            redoStack.push(inverse);  // Actually the inverse of inverse = original
            broadcast(inverse);
        }
    }
    
    private Verb generateInverse(Verb verb) {
        switch(verb.action) {
            case "CREATE_WALL":
                // Inverse is DELETE_WALL
                return new Verb("DELETE_WALL", verb.target, null, 
                               verb.user, System.currentTimeMillis());
                
            case "APPLY_MATERIAL":
                // Need to know previous material
                String previous = db.getPreviousMaterial(verb.target);
                return new Verb("APPLY_MATERIAL", verb.target, 
                               Map.of("material", previous),
                               verb.user, System.currentTimeMillis());
                
            case "MOVE_ELEMENT":
                // Need previous position
                Point3D previousPos = db.getPreviousPosition(verb.target);
                return new Verb("MOVE_ELEMENT", verb.target,
                               Map.of("position", previousPos),
                               verb.user, System.currentTimeMillis());
        }
    }
}
```
