## Background: Why DXF matters

SVG is a browser sidecar. DXF is the professional deliverable:
- Opens in AutoCAD, ArchiCAD, FreeCAD, QCAD without conversion
- Coordinates in model-space mm → W-2D-ROUNDTRIP overlay diff is direct
- AIA layer names → architects freeze/thaw layers as expected
- LTSCALE + DIMSTYLE → annotations auto-scale at any scale

### Setup (from §14.5):
```python
doc = ezdxf.new('R2010')
doc.header['$INSUNITS'] = 4          # mm
doc.header['$LTSCALE']  = scale / 100.0
msp = doc.modelspace()

# AIA layers (§14.2)
layers = [
    ('A-WALL-FULL', 7, 'CONTINUOUS'),
    ('A-WALL-PRTN', 7, 'CONTINUOUS'),
    ('A-WALL-PATT', 254, 'CONTINUOUS'),
    ('A-GLAZ',      5,  'CONTINUOUS'),
    ('A-DOOR',      3,  'CONTINUOUS'),
    ('A-GRID',      8,  'CENTER'),
    ('A-ANNO-DIMS', 7,  'CONTINUOUS'),
    ('A-ANNO-TEXT', 7,  'CONTINUOUS'),
    ('A-FURN',      9,  'CONTINUOUS'),
]
for name, color, ltype in layers:
    doc.layers.add(name=name, color=color, linetype=ltype)

# DIMSTYLE (§14.4): ARCHTICK, DIMSCALE, text height
dimstyle = doc.dimstyles.new('ARCH_JKR')
dimstyle.dxf.dimscale  = scale
dimstyle.dxf.dimtxt    = 2.5    # 2.5mm on paper = 250 in model at 1:100
dimstyle.dxf.dimblk    = 'ARCHTICK'
dimstyle.dxf.dimtxsty  = 'Standard'
```

### Geometry — model-space mm (no PAPER_FACTOR):
```python
# CUT elements: section contours → LWPOLYLINE on A-WALL-FULL / A-WALL-PRTN
# Each contour point: (x_world_m * 1000, y_world_m * 1000) in mm
for es in cut_elements:
    layer = 'A-WALL-FULL' if es.ifc_class == 'IfcWall' and 'Ext' in es.element_name \
            else 'A-WALL-PRTN'
    for contour in es.contours:
        pts = [(p[0] * 1000, p[1] * 1000) for p in contour.points]
        msp.add_lwpolyline(pts, close=True, dxfattribs={'layer': layer,
                            'lineweight': 50})  # 0.50mm = DXF lineweight 50

# BELOW elements: bounding box rect on A-FURN
for es in below_elements:
    bx0, by0, bx1, by1 = [v * 1000 for v in es.bbox_2d]
    msp.add_lwpolyline([(bx0,by0),(bx1,by0),(bx1,by1),(bx0,by1)],
                       close=True, dxfattribs={'layer': 'A-FURN', 'lineweight': 18})

# Grid lines: CENTER linetype, A-GRID layer, extend beyond building
for g in grids:
    if g.axis == 'x':
        msp.add_line((g.position*1000, y_min-500), (g.position*1000, y_max+500),
                     dxfattribs={'layer': 'A-GRID', 'lineweight': 18})
        # Bubble: circle + text
        msp.add_circle((g.position*1000, y_max+700), 400,
                        dxfattribs={'layer': 'A-GRID'})
        msp.add_text(g.label, dxfattribs={'layer': 'A-ANNO-TEXT',
                     'height': 2.5*scale}).set_placement((g.position*1000, y_max+700))
    # (similar for y-axis grids)
```
