-- Tag vocabulary table (governance)
CREATE TABLE tag_vocabulary (
    tag TEXT PRIMARY KEY,
    category TEXT NOT NULL,  -- USAGE, SOURCE, DISCIPLINE
    description TEXT
);

-- Canonical tags
INSERT INTO tag_vocabulary VALUES
    ('RESIDENTIAL', 'USAGE', 'Suitable for residential construction'),
    ('COMMERCIAL', 'USAGE', 'Suitable for commercial construction'),
    ('COMMERCIAL_ONLY', 'USAGE', 'Commercial only - not for residential'),
    ('SOURCE:TERMINAL', 'SOURCE', 'Extracted from TERMINAL lobby IFC'),
    ('SOURCE:IFC_SAMPLE', 'SOURCE', 'Extracted from IFC research samples'),
    ('SOURCE:RESEARCHED', 'SOURCE', 'From standards/specs, not extracted');

-- Component tags with foreign key governance
CREATE TABLE component_tags (
    component_id INTEGER REFERENCES component_definitions(id),
    tag TEXT REFERENCES tag_vocabulary(tag),
    PRIMARY KEY (component_id, tag)
);

CREATE INDEX idx_component_tags ON component_tags(tag);

-- Population: MEP fixtures (plumbing) - RESIDENTIAL
INSERT INTO component_tags (component_id, tag)
SELECT cd.id, 'RESIDENTIAL'
FROM component_definitions cd
JOIN component_types ct ON cd.type_id = ct.id
WHERE ct.ifc_class IN (
    'IfcFlowTerminal',      -- Fixtures (WC, basin, shower)
    'IfcPipeFitting',       -- Elbows, tees, couplings
    'IfcValve'              -- Isolation valves
);

-- Railings - RESIDENTIAL (architectural, not MEP)
INSERT INTO component_tags (component_id, tag)
SELECT cd.id, 'RESIDENTIAL'
FROM component_definitions cd
JOIN component_types ct ON cd.type_id = ct.id
WHERE ct.ifc_class = 'IfcRailing';

-- Single-leaf doors only (< 1.0m width)
INSERT INTO component_tags (component_id, tag)
SELECT cd.id, 'RESIDENTIAL'
FROM component_definitions cd
JOIN component_types ct ON cd.type_id = ct.id
WHERE ct.ifc_class = 'IfcDoor'
AND (cd.local_max_y - cd.local_min_y) < 1.0;

-- Commercial-only items
INSERT INTO component_tags (component_id, tag)
SELECT cd.id, 'COMMERCIAL_ONLY'
FROM component_definitions cd
JOIN component_types ct ON cd.type_id = ct.id
WHERE ct.ifc_class IN (
    'IfcFireSuppressionTerminal',  -- Sprinklers
    'IfcAirTerminal'               -- Commercial HVAC diffusers
);

-- Source tracking for all TERMINAL components
INSERT INTO component_tags (component_id, tag)
SELECT id, 'SOURCE:TERMINAL'
FROM component_definitions;
